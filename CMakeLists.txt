cmake_minimum_required(VERSION 3.24)
project(cuda_kernel_performance_lab LANGUAGES CXX)

option(CUDA_BACKEND "Build CUDA kernels and the GPU benchmark" ON)
option(ENABLE_NVBENCH "Fetch NVBench and build parameter-sweep benchmarks" OFF)
include(CTest)
add_library(kernel_reference src/reference.cpp)
target_include_directories(kernel_reference PUBLIC include)
target_compile_features(kernel_reference PUBLIC cxx_std_17)

add_executable(reference_tests tests/reference_tests.cpp)
target_link_libraries(reference_tests PRIVATE kernel_reference)
add_test(NAME cpu_reference_tests COMMAND reference_tests)

if(CUDA_BACKEND)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    add_executable(cuda_benchmark src/cuda_benchmark.cu src/cuda_kernels.cu)
    target_link_libraries(cuda_benchmark PRIVATE kernel_reference)
    target_include_directories(cuda_benchmark PRIVATE include)
    set_target_properties(cuda_benchmark PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    if(ENABLE_NVBENCH)
      if(CMAKE_VERSION VERSION_LESS 3.30.4)
        message(FATAL_ERROR "NVBench requires CMake 3.30.4 or newer")
      endif()
      include(FetchContent)
      set(NVBench_ENABLE_CUPTI OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_NVML OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_HEADER_TESTING OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_EXECUTABLES OFF CACHE BOOL "" FORCE)
      set(NVBench_ENABLE_INSTALL_RULES OFF CACHE BOOL "" FORCE)
      FetchContent_Declare(nvbench
        GIT_REPOSITORY https://github.com/NVIDIA/nvbench.git
        GIT_TAG python-0.3.0
        GIT_SHALLOW TRUE)
      FetchContent_MakeAvailable(nvbench)
      add_executable(nvbench_kernels src/nvbench_kernels.cu src/cuda_kernels.cu)
      target_link_libraries(nvbench_kernels PRIVATE nvbench::main)
      target_include_directories(nvbench_kernels PRIVATE include)
      set_target_properties(nvbench_kernels PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    endif()
  else()
    if(ENABLE_NVBENCH)
      message(FATAL_ERROR "ENABLE_NVBENCH requires a CUDA compiler; install CUDA Toolkit or set CUDA_BACKEND=OFF")
    endif()
    message(STATUS "CUDA compiler not found; building CPU reference tests only")
  endif()
endif()
